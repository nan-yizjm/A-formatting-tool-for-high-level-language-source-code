/* program_error.txt —— 语法分析错误清单
 * 语法分析和词法分析不一样，碰到第一个错误就停，所以整个文件跑起来只会报第一个。
 * 想单独看某个错误，把它那段复制到新文件再跑就行。
 * 报错分两种：语义/类型检查会带上“错误原因：xxx”，结构文法错误只报“发生生成错误”加行号。 */

/* ===== 一、类型和变量声明 ===== */

/* 01 类型说明符组合不合法，会报“非法的类型说明符组合” */
int char x;

/* 02 类型后面忘了写变量名，只报发生生成错误 */
int ;

/* 03 同一个作用域里变量重复定义 */
int x;
int x;

/* 04 变量名和函数名撞了（先变量后函数、先函数后变量都会报） */
int f;
int f() { return 0; }

/* 05 等号后面没有初始化表达式 */
int x = ;

/* 06 变量声明少了分号或逗号 */
int x y;

/* 07 数组下标必须是整型常量，不能是变量 */
int a[x];

/* 08 数组下标少了右中括号 */
int a[3;

/* ===== 二、函数 ===== */

/* 09 形参名重复了 */
int f(int a, int a) { return a; }

/* 10 函数重复定义（先写原型再定义不算重复） */
int f() { return 0; }
int f() { return 0; }

/* 11 调用了一个没声明过的函数 */
int f() { return g(); }

/* 12 实参个数和形参对不上 */
int f(int a) { return a; }
int g() { return f(); }

/* 13 实参类型和形参对不上 */
int f(int a) { return a; }
int g() { return f(1.0); }

/* 14 形参列表括号里的东西不合法 */
int f(, int a) { return a; }

/* 15 写了 void 形参后面还跟了别的 */
int f(void int a) { return 0; }

/* 16 函数声明/定义后面少了分号或左大括号 */
int f() = 0;

/* 17 形参后面少了右括号或逗号 */
int f(int a int b) { return a; }

/* 18 形参类型后面少了形参名 */
int f(int) { return 0; }

/* ===== 三、复合语句 ===== */

/* 19 复合语句少了右大括号 */
int f() { return 0;

/* ===== 四、选择语句 ===== */

/* 20 if 后面少了左括号 */
int f() { if a) return 0; return 0; }

/* 21 if 的条件不能为空 */
int f() { if () return 0; return 0; }

/* ===== 五、循环语句 ===== */

/* 22 for 后面少了左括号 */
int f() { for ; return 0; }

/* 23 for 的终止条件不能为空（初始表达式和循环表达式可以空） */
int f() { int i; for (i = 0; ; i = i + 1) return 0; }

/* 24 while 后面少了左括号 */
int f() { while a) return 0; return 0; }

/* 25 while 的条件不能为空 */
int f() { while () return 0; return 0; }

/* ===== 六、跳转语句 ===== */

/* 26 continue 后面少了分号 */
int f() { continue 0; }

/* 27 break 后面少了分号 */
int f() { break 0; }

/* 28 return 必须带一个表达式，不能光写 return; */
int f() { return; }

/* 29 return 的表达式类型和函数返回类型不一致 */
int f() { return 1.0; }

/* ===== 七、表达式语句 ===== */

/* 30 用了一个没声明过的变量 */
int f() { return x; }

/* 31 数组不能整体参与赋值，只能按下标来 */
int a[3];
int f() { a = 1; return 0; }

/* 32 const 对象不能被赋值 */
const int a = 5;
int f() { a = 6; return 0; }

/* 33 赋值等号左边必须是能修改的东西 */
int f() { 5 = 3; return 0; }

/* 34 运算符两边的类型不兼容 */
int a[3];
int f() { return a + 1; }

/* 35 取模要求两边都是整数 */
int f() { return 1.0 % 2; }

/* 36 表达式语句少了分号 */
int a;
int f() { a = 1 }

/* ===== 八、没支持的关键字 ===== */

/* 37 switch / do / goto / case / default 这些还没实现 */
int f() { goto x; }

/* 另外 program.txt 里刻意避开的写法：指针、数组形参、一元运算、 */
/* sizeof、逗号表达式、花括号初始化 */
