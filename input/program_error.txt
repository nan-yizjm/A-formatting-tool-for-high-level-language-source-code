/* ===== 一、类型和变量声明 ===== */

/* 01 类型说明符组合不合法 */
int char x1;

/* 02 类型后面忘了写变量名 */
int ;

/* 03 同一个作用域里变量重复定义 */
int x3;
int x3;

/* 04 变量名和函数名重复 */
int f4;
int f4() { return 0; }

/* 05 等号后面没有初始化表达式 */
int x5 = ;

/* 06 变量声明少了分号或逗号 */
int x6 y6;

/* 07 数组下标必须是整型常量，不能是变量 */
int a7[x7];

/* 08 数组下标少了右中括号 */
int a8[3;


/* ===== 二、函数 ===== */

/* 09 形参名重 */
int f9(int a9, int a9) { return a9; }

/* 10 函数重复定义 */
int f10() { return 0; }

/* 11 调用未声明过的函数 */
int f11() { return g11(); }

/* 12 实参个数和形参不匹配  */
int f12(int a12) { return a12; }
int g12() { return f12(); }

/* 13 实参类型和形参对不上 */
int f13(int a13) { return a13; }
int g13() { return f13(1.0); }

/* 16 函数声明/定义后面少了分号或左大括号 */
int f16() = 0;

/* 17 形参后面少了右括号或逗号 */
int f17(int a17 int b17) { return a17; }

/* ===== 四、选择语句 ===== */

/* 20 if 后面少了左括号 */
int f20() { if a20) return 0; return 0; }

/* 21 if 的条件不能为空 */
int f21() { if () return 0; return 0; }


/* ===== 五、循环语句 ===== */

/* 22 for 后面少了左括号 */
int f22() { for ; return 0; }

/* 23 for 的终止条件不能为空（初始表达式和循环表达式可以空） */
int f23() { int i23; for (i23 = 0; ; i23 = i23 + 1) return 0; }

/* 24 while 后面少了左括号 */
int f24() { while a24) return 0; return 0; }

/* 25 while 的条件不能为空 */
int f25() { while () return 0; return 0; }


/* ===== 六、跳转语句 ===== */

/* 29 return 的表达式类型和函数返回类型不一致 */
int f29() { return 1.0; }


/* ===== 七、表达式语句 ===== */

/* 30 用了一个没声明过的变量 */
int f30() { return x30; }

/* 31 数组不能整体参与赋值，只能按下标来 */
int a31[3];
int f31() { a31 = 1; return 0; }

/* 32 const 对象不能被赋值 */
const int a32 = 5;
int f32() { a32 = 6; return 0; }

/* 33 赋值等号左边必须是能修改的东西 */
int f33() { 5 = 3; return 0; }

/* 34 运算符两边的类型不兼容 */
int a34[3];
int f34() { return a34 + 1; }

/* 35 取模要求两边都是整数 */
int f35() { return 1.0 % 2; }

/* 36 表达式语句少了分号 */
